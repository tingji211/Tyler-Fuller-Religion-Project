run_multinom_model <- function(data, outcome_var, predictor_vars, ref_level = "In-person only") {
  library(nnet)
  library(dplyr)
  
  # Step 1: remove 99
  data_clean <- data %>%
    filter(if_all(all_of(predictor_vars), ~ . != 99))

  # Step 2: convert predictors to factor
  data_clean <- data_clean %>%
    mutate(across(all_of(predictor_vars), as.factor))

  # Step 3: set reference category for outcome
  data_clean[[outcome_var]] <- relevel(as.factor(data_clean[[outcome_var]]), ref = ref_level)

  # Step 4: drop predictors with only 1 level
  level_counts <- sapply(data_clean[, predictor_vars, drop = FALSE], nlevels)
  invalid_vars <- names(level_counts[level_counts <= 1])

  if (length(invalid_vars) > 0) {
    message("⚠️ Dropped variables (only 1 level): ",
            paste(invalid_vars, collapse = ", "))
  }

  valid_vars <- setdiff(predictor_vars, invalid_vars)

  # Step 5: build formula
  formula_obj <- as.formula(
    paste(outcome_var, "~", paste(valid_vars, collapse = " + "))
  )

  # Step 6: fit multinom model
  model <- multinom(formula_obj, data = data_clean)

  # Step 7: return model & summary
  list(
    model = model,
    summary = summary(model),
    valid_vars = valid_vars,
    dropped_vars = invalid_vars
  )
}
